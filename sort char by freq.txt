class Solution {
public:
    string frequencySort(string s) {
        unordered_map<char, int> mp;
        for(char c : s){
            mp[c]++;
        }
        vector <pair<char , int>> vtr(mp.begin(),mp.end());
        sort(vtr.begin(),vtr.end(),[](auto &a , auto &b){
            return a.second>b.second;
        });
        string str="";
        // for(int i=0;i<vtr.size();i++){
        //     for(int j=0 ; j<vtr[i].second;j++){
        //         str+=vtr[i].first;
        //     }
        // }
        for(auto it:vtr){
            for(int i=0;i<it.second;i++){
            str+=it.first;
            }
        }
        return str;
    }
};
