class Solution {
public:
    int maxDepth(string s) {
        stack<char> st;
        int count = 0;

        for(char c : s) {

            if(c == '(') {
                st.push(c);
                count = max(count, (int)st.size());
            }
            else if(c == ')') {
                st.pop();
            }
        }

        return count;
    }
};
