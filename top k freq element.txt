class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        unordered_map<int, int> mp;
        for(int x: nums){
            mp[x]++;
        }
        vector <pair<int,int>> vc(mp.begin(), mp.end());
        sort(vc.begin(),vc.end(),[](auto &a , auto&b){
            return a.second>b.second;
        });
        vector <int> vr;

        for(int i=0 ; i<k;i++){
            vr.push_back(vc[i].first);
        }

        return vr;

    }
};
